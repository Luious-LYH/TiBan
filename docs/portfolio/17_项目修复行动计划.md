```python
from datetime import datetime, timedelta
from dataclasses import dataclass
from typing import Literal

Quality = Literal[0, 1, 2, 3, 4, 5]

@dataclass
class FSRSCard:
    """FSRS 复习卡片"""
    question_id: str
    due: datetime
    stability: float  # 记忆稳定性（天）
    difficulty: float  # 难度系数（1-10）
    elapsed_days: int  # 距离上次复习的天数
    scheduled_days: int  # 本次计划间隔
    reps: int  # 重复次数
    lapses: int  # 遗忘次数
    state: Literal["new", "learning", "review", "relearning"]
    last_review: datetime | None

class FSRSScheduler:
    """FSRS-4.5 调度器"""
    
    # 参数（来自 FSRS 论文）
    w = [0.4, 0.6, 2.4, 5.8, 4.93, 0.94, 0.86, 0.01, 1.49, 0.14, 0.94, 2.18, 0.05, 0.34, 1.26, 0.29, 2.61]
    
    def __init__(self):
        self.request_retention = 0.9  # 目标保持率
        self.maximum_interval = 36500  # 最大间隔（100 年）
        self.easy_bonus = 1.3
        self.hard_interval = 1.2
    
    def init_card(self, question_id: str) -> FSRSCard:
        """初始化新卡片"""
        return FSRSCard(
            question_id=question_id,
            due=datetime.now(),
            stability=0,
            difficulty=0,
            elapsed_days=0,
            scheduled_days=0,
            reps=0,
            lapses=0,
            state="new",
            last_review=None
        )
    
    def review(self, card: FSRSCard, rating: Quality) -> FSRSCard:
        """复习并更新卡片"""
        now = datetime.now()
        
        if card.last_review:
            card.elapsed_days = (now - card.last_review).days
        
        # 计算新的稳定性和难度
        if card.state == "new":
            card.difficulty = self._init_difficulty(rating)
            card.stability = self._init_stability(rating)
            card.state = "learning"
        else:
            card.difficulty = self._next_difficulty(card, rating)
            card.stability = self._next_stability(card, rating)
        
        # 计算下次复习间隔
        card.scheduled_days = self._next_interval(card.stability)
        card.due = now + timedelta(days=card.scheduled_days)
        
        # 更新状态
        if rating < 3:  # Again / Hard
            card.lapses += 1
            card.state = "relearning"
        else:
            card.reps += 1
            card.state = "review"
        
        card.last_review = now
        return card
    
    def _init_stability(self, rating: Quality) -> float:
        """初始稳定性"""
        return max(self.w[rating - 1], 0.1)
    
    def _init_difficulty(self, rating: Quality) -> float:
        """初始难度"""
        return min(max(self.w[4] - (rating - 3) * self.w[5], 1), 10)
    
    def _next_difficulty(self, card: FSRSCard, rating: Quality) -> float:
        """下次难度"""
        delta_d = -self.w[6] * (rating - 3)
        new_d = card.difficulty + delta_d
        return min(max(self._mean_reversion(card.difficulty, new_d), 1), 10)
    
    def _next_stability(self, card: FSRSCard, rating: Quality) -> float:
        """下次稳定性"""
        if rating > 1:
            # 记住
            new_s = card.stability * (
                1 + 
                math.exp(self.w[8]) * 
                (11 - card.difficulty) * 
                math.pow(card.stability, -self.w[9]) * 
                (math.exp((1 - card.elapsed_days / card.stability) * self.w[10]) - 1)
            )
        else:
            # 遗忘
            new_s = (
                self.w[11] * 
                math.pow(card.difficulty, -self.w[12]) * 
                (math.pow(card.stability + 1, self.w[13]) - 1) * 
                math.exp((1 - card.elapsed_days / card.stability) * self.w[14])
            )
        return max(new_s, 0.1)
    
    def _next_interval(self, stability: float) -> int:
        """下次间隔"""
        interval = stability * (math.log(self.request_retention) / math.log(0.9))
        return min(max(round(interval), 1), self.maximum_interval)
    
    def _mean_reversion(self, current: float, new: float) -> float:
        """均值回归"""
        return self.w[7] * self.w[4] + (1 - self.w[7]) * new

# 集成到 memory_service
class MemoryService:
    def __init__(self):
        self.fsrs = FSRSScheduler()
        self.cards: dict[str, FSRSCard] = {}
    
    def review_question(self, question_id: str, quality: Quality):
        """复习题目"""
        if question_id not in self.cards:
            self.cards[question_id] = self.fsrs.init_card(question_id)
        
        card = self.cards[question_id]
        updated_card = self.fsrs.review(card, quality)
        self.cards[question_id] = updated_card
        
        return {
            "question_id": question_id,
            "next_review": updated_card.due.isoformat(),
            "interval_days": updated_card.scheduled_days,
            "difficulty": round(updated_card.difficulty, 2),
            "stability": round(updated_card.stability, 2),
        }
    
    def get_due_cards(self) -> list[FSRSCard]:
        """获取到期复习的卡片"""
        now = datetime.now()
        return [card for card in self.cards.values() if card.due <= now]
```

```python
# backend/app/services/rag_service.py
import chromadb
from sentence_transformers import SentenceTransformer
from typing import Literal

class MedicalRAG:
    def __init__(self):
        # 中文医学预训练模型
        self.encoder = SentenceTransformer('shibing624/text2vec-base-chinese-paraphrase')
        
        # ChromaDB 客户端
        self.client = chromadb.Client()
        self.collection = self.client.create_collection(
            name="medical_knowledge",
            metadata={"hnsw:space": "cosine"}
        )
        
        # 加载知识库
        self._load_knowledge()
    
    def _load_knowledge(self):
        """加载医学知识到向量库"""
        docs = read_json("real_sample_knowledge.json")
        
        for doc in docs:
            # 分块
            chunks = self.chunk_document(doc["content"])
            
            for i, chunk in enumerate(chunks):
                chunk_id = f"{doc['id']}_chunk_{i}"
                embedding = self.encoder.encode([chunk])[0].tolist()
                
                self.collection.add(
                    ids=[chunk_id],
                    embeddings=[embedding],
                    documents=[chunk],
                    metadatas=[{
                        "doc_id": doc["id"],
                        "body_part": doc.get("body_part", ""),
                        "chunk_index": i,
                        "source": doc.get("source", "")
                    }]
                )
    
    def chunk_document(self, text: str, max_len: int = 200) -> list[str]:
        """按句子分块"""
        sentences = text.split('。')
        chunks = []
        current_chunk = ""
        
        for sent in sentences:
            if len(current_chunk) + len(sent) < max_len:
                current_chunk += sent + "。"
            else:
                if current_chunk:
                    chunks.append(current_chunk)
                current_chunk = sent + "。"
        
        if current_chunk:
            chunks.append(current_chunk)
        
        return chunks
    
    def hybrid_search(
        self,
        query: str,
        body_part: str | None = None,
        top_k: int = 5,
        alpha: float = 0.3  # BM25 权重
    ) -> list[dict]:
        """混合检索：alpha * BM25 + (1 - alpha) * Dense"""
        
        # Dense 检索（向量相似度）
        query_vec = self.encoder.encode([query])[0].tolist()
        
        where = {"body_part": body_part} if body_part else None
        dense_results = self.collection.query(
            query_embeddings=[query_vec],
            n_results=top_k * 2,  # 多检索一些用于融合
            where=where
        )
        
        # BM25 检索（已有实现）
        sparse_results = self._bm25_search(query, body_part, top_k * 2)
        
        # RRF (Reciprocal Rank Fusion) 融合
        return self._rrf_merge(
            dense_results,
            sparse_results,
            k=60,
            top_k=top_k
        )
    
    def _rrf_merge(
        self,
        dense_results: dict,
        sparse_results: list[dict],
        k: int = 60,
        top_k: int = 5
    ) -> list[dict]:
        """RRF 融合"""
        scores = {}
        
        # Dense 结果打分
        for rank, doc_id in enumerate(dense_results["ids"][0]):
            scores[doc_id] = scores.get(doc_id, 0) + 1 / (k + rank + 1)
        
        # Sparse 结果打分
        for rank, doc in enumerate(sparse_results):
            doc_id = doc["id"]
            scores[doc_id] = scores.get(doc_id, 0) + 1 / (k + rank + 1)
        
        # 排序并返回 top_k
        sorted_docs = sorted(scores.items(), key=lambda x: x[1], reverse=True)[:top_k]
        
        return [
            {
                "id": doc_id,
                "score": score,
                "content": self._get_doc_content(doc_id),
                "metadata": self._get_doc_metadata(doc_id)
            }
            for doc_id, score in sorted_docs
        ]
```

```python
# backend/app/services/agent_runtime_v2.py
from openai import OpenAI

class AgentRuntimeV2:
    """基于 OpenAI Function Calling 的 Agent Runtime"""
    
    def __init__(self):
        self.client = OpenAI()
        self.tools = self._define_tools()
    
    def _define_tools(self) -> list[dict]:
        """定义工具"""
        return [
            {
                "type": "function",
                "function": {
                    "name": "retrieve_evidence",
                    "description": "从医学知识库中检索相关证据",
                    "parameters": {
                        "type": "object",
                        "properties": {
                            "query": {
                                "type": "string",
                                "description": "检索查询，应包含关键医学术语"
                            },
                            "body_part": {
                                "type": "string",
                                "enum": ["食管", "胃", "小肠", "全部"],
                                "description": "限制检索部位"
                            },
                            "top_k": {
                                "type": "integer",
                                "description": "返回结果数量",
                                "default": 5
                            }
                        },
                        "required": ["query"]
                    }
                }
            },
            {
                "type": "function",
                "function": {
                    "name": "grade_answer",
                    "description": "根据 Rubric 评分学生答案",
                    "parameters": {
                        "type": "object",
                        "properties": {
                            "student_answer": {
                                "type": "string",
                                "description": "学生的答案"
                            },
                            "rubric": {
                                "type": "array",
                                "items": {"type": "string"},
                                "description": "评分标准（原子事实列表）"
                            },
                            "evidence": {
                                "type": "array",
                                "items": {"type": "object"},
                                "description": "检索到的证据"
                            }
                        },
                        "required": ["student_answer", "rubric"]
                    }
                }
            },
            {
                "type": "function",
                "function": {
                    "name": "safety_check",
                    "description": "检查答案是否越界诊断",
                    "parameters": {
                        "type": "object",
                        "properties": {
                            "answer": {
                                "type": "string",
                                "description": "待检查的答案"
                            }
                        },
                        "required": ["answer"]
                    }
                }
            }
        ]
    
    async def run(
        self,
        case_id: str,
        learner_answer: str,
        **kwargs
    ) -> dict:
        """执行 Agent 工作流"""
        case = self.get_case(case_id)
        messages = [
            {
                "role": "system",
                "content": "你是一个医学教学评分 Agent。根据学生答案，调用工具完成评分。"
            },
            {
                "role": "user",
                "content": f"题目：{case['question']}\n学生答案：{learner_answer}\n请评分。"
            }
        ]
        
        max_iterations = 5
        tool_calls_history = []
        
        for i in range(max_iterations):
            # 调用 LLM
            response = await self.client.chat.completions.create(
                model="gpt-4o",
                messages=messages,
                tools=self.tools,
                tool_choice="auto"
            )
            
            message = response.choices[0].message
            messages.append(message)
            
            # 如果没有 tool_calls，结束
            if not message.tool_calls:
                break
            
            # 执行工具调用
            for tool_call in message.tool_calls:
                function_name = tool_call.function.name
                function_args = json.loads(tool_call.function.arguments)
                
                # 调用真实工具函数
                if function_name == "retrieve_evidence":
                    result = await self._retrieve_evidence(**function_args)
                elif function_name == "grade_answer":
                    result = await self._grade_answer(**function_args)
                elif function_name == "safety_check":
                    result = await self._safety_check(**function_args)
                else:
                    result = {"error": "Unknown function"}
                
                # 记录工具调用
                tool_calls_history.append({
                    "function": function_name,
                    "arguments": function_args,
                    "result": result
                })
                
                # 将结果返回给 LLM
                messages.append({
                    "role": "tool",
                    "tool_call_id": tool_call.id,
                    "name": function_name,
                    "content": json.dumps(result, ensure_ascii=False)
                })
        
        return {
            "final_response": message.content,
            "tool_calls": tool_calls_history,
            "iterations": i + 1
        }
```
